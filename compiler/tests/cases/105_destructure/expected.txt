MODULE
  func divide
    param a: Int
    param b: Int
    return result: Int
    return error: opt Error
    body
      if (b is 0)
        then
          ret result: 0, error: "divide by zero"
      ret result: (a / b), error: none
  func useDivide
    param a: Int
    param b: Int
    return result: Int
    body
      destructure
        let sum: Int <- result
        let err: opt Error <- error
        call divide(a: a, b: b)
      ret sum
