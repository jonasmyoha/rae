func loops(items: view List(Int)) {
  var whileIndex: Int = 0
  loop whileIndex < items.length {
    whileIndex++
  }
  loop var prefixIndex: Int = 0, prefixIndex < items.length, ++prefixIndex {
    log(prefixIndex)
  }
  loop var postfixIndex: Int = 0, postfixIndex < items.length, postfixIndex++ {
    log(postfixIndex)
  }
  loop let item: Int in items {
    log(item)
  }
  loop let item: view Int in items {
    log(item)
  }
  loop {
    break
  }
}
